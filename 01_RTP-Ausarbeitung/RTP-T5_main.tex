    \chapter{Preparations}\label{chap:preperations}

        \section{(aV)  P controller in the time domain}\label{sec:a-p-controller-time-domain}

            \begin{displaycquote}{RTP_Versuche}
                \footnotesize{\textit{(aV) What is the equation that describes a digital P controller in the time domain?}}
            \end{displaycquote}\label{task(aV)}

            \begin{equation}
                x_a(t) = K_P \cdot x_e(t)
            \end{equation}

            \begin{enumerate}
                \item $x_e(t)$: Input variable of the P controller
                \item $K_p$: Proportional coefficient
                \item $x_a(t)$: Control output of the P controller
            \end{enumerate}


        \section{(bV) Transfer function of open loop P controller}\label{sec:b-transfer-function-open-loop-p-controller}
            
            \begin{displaycquote}{RTP_Versuche}
                \footnotesize{\textit{(bV) What is the transfer function of the open loop with P controller?\\
                Which characteristic shows $G_o(s)$ ?}}
            \end{displaycquote}\label{task(bV)}

            The transfer function $G_o(s)$ is obtained using the block diagram determined in the fourth experiment as follows:
            \begin{equation}
                G_o(s) = K_P\cdot \frac{-K_I}	{T_1\cdot s^2 + s }
            \end{equation}\label{eq:G_o(s)}

            $G_o(s)$ exhibits IT1 behavior.

        \section{(cV) Transfer function of distance control with P controller}\label{sec:c-transfer-function-distance-control-p-controller}

            \begin{displaycquote}{RTP_Versuche}
                \footnotesize{\textit{(cV) What is the transfer function of the distance control with P controller?\\
                Which characteristic shows $G_W(s)$ ?}}
            \end{displaycquote}\label{task(cV)}

            By inserting \ref{eq:G_o(s)}, we obtain the following for $G_W(s)$:

            \begin{equation}
                G_W(s) = \frac{G_o(s)}{1 + G_o(s)} = \frac{1}{1 + \frac{-T_1}{K_P\cdot K_I}  \cdot s^2 - \frac{1}{K_P\cdot K_I}\cdot s}
            \end{equation}\label{eq:G_W(s)}  

            The closed-loop system $G_W(s)$ exhibits $PT_2$ behavior.


        \section{(dV) Reasonable damping ratio D for the closed control loop}\label{sec:d-reasonable-damping-ratio}

            \begin{displaycquote}{RTP_Versuche}
                \footnotesize{\textit{(dV) Select a reasonable damping ratio D for the closed control loop and calculate the controller parameter $K_P$ so that the desired/selected damping ratio is achieved.}}
            \end{displaycquote}\label{task(dV)}
            
            The damping ratio is set to D = 1 (aperiodic limit case) to ensure a fast, overshoot-free settling. This allows the robot to reach the target distance as quickly as possible without having to move backward. The standard mathematical form of a PT2 system is:

            \begin{equation}
                G_s = \frac{K_P}{\frac{1}{\omega _0^2} \cdot s^2 + \frac{2D}{\omega _0 \cdot s} + 1}
            \end{equation}\label{eq:PT_2}

            Rearranging \ref{eq:PT_2} yields:

            \begin{equation}
                \frac{1}{\omega _0 ^2} = - \frac{T_1}{K_P \cdot K_I} \rightarrow \omega_0 = \sqrt{-\frac{K_P \cdot K_I}{T_1}}
                \label{eq:w_0}
            \end{equation}

            \begin{equation}
                \frac{2D}{\omega _0 } = - \frac{1}{K_P \cdot K_I} \rightarrow \omega_0 = - K_P \cdot K_I \cdot 2 \cdot D
                \label{eq:2Dw_0}
            \end{equation}
            
            If we now set Equation \ref{eq:2Dw_0} equal to \ref{eq:w_0}, we can solve for $K_P$.

            \begin{equation}
                \sqrt{-\frac{K_P \cdot K_I}{T_1}} = - K_P \cdot K_I \cdot 2 \cdot D \rightarrow K_P = \frac{-1}{4 \cdot D^2 \cdot T_1\cdot K_I}
            \end{equation}\label{eq:K_P}

            Using the previously determined parameters (mean value) $K_I$ and $T_1$, 

            Left motor:
            \begin{equation}
                K_{I,L} = \qty{0.437}{\centi\metre\per\second\per\percent}, \qquad
                T_{1,L} = \qty{57.1}{\milli\second} = \qty{0.0571}{\second}
            \end{equation}

            Right motor:
            \begin{equation}
                K_{I,R} = \qty{0.446}{\centi\metre\per\second\per\percent}, \qquad
                T_{1,R} = \qty{52.2}{\milli\second} = \qty{0.0522}{\second}
            \end{equation}

            the following values are obtained:

            Left motor:
            \begin{equation}
                K_{P,L} = \frac{-1}{4 \cdot 1^2 \cdot 0.0571 \cdot 0.437} = \qty{-9.99}{\percent\per\cm} \approx \qty{-10}{\percent\per\cm}
            \end{equation}

            Right motor:
            \begin{equation}
                K_{P,R} = \frac{-1}{4 \cdot 1^2 \cdot 0.0522 \cdot 0.446} = \qty{-10.77}{\percent\per\cm} \approx \qty{-11}{\percent\per\cm}
            \end{equation}

        \section{f) Reference, manipulated and controlled variable}\label{sec:f-}

            \begin{displaycquote}{RTP_Versuche}
                \footnotesize{\textit{(f) Plot the reference variable, the manipulated variable, and the controlled variable via Matlab (clean graphs, no screenshots from the scope). Is the damping level you selected correct? Interpret the signal curves.}}
            \end{displaycquote}\label{task(f)}


            \begin{figure}[H]
                \centering
                \includegraphics[width=\textwidth]{figure/ContainerScopePrintToFigure.pdf}
                \caption{Simulated step response of the closed control loop for $D=1$: reference variable $w$, input variable $x_a$ (saturated to $\pm\qty{100}{\percent}$), and controlled variable $x$.}
                \label{fig:container-scope}
            \end{figure}

            \begin{itemize}
                \item \textbf{1. Evaluation of the damping:} The selected damping proves to be well-suited. The simulation shows that the controlled variable $x$ (actual distance) rapidly converges to the reference variable $w$ (setpoint distance) without significant overshoot or instability. The system exhibits the desired aperiodic transient response, which is ideal for distance control to avoid collisions.
                \item \textbf{2. Interpretation of the Signal Curves:} 
                \begin{itemize}
                    \item \textbf{Reference variable $\mathbf{w(t)}$:} It represents a step change from \qty{80}{\cm} to \qty{25}{\cm}.
                    
                    \item \textbf{Manipulated variable $\mathbf{x_a(t)}$:} At the start, the control error is at its maximum. The P controller theoretically calculates an extremely high manipulated variable from this. However, since the actual EV3 driver is limited to $\pm\qty{100}{\percent}$, the motor moves to its initial full saturation ("full throttle"). Only damped by the rate limiter, the data shows that the manipulated variable $x_a$ does not drop significantly until shortly before reaching the target. This marks the transition from saturation to the linear control region, in order to bring the robot to a stop at the target value.
                    
                    \item \textbf{Controlled variable $\mathbf{x(t)}$:} The distance initially decreases quadratically then it transitions to a nearly linear stretch before slowing in the final approach, since the motor runs at a constant maximum speed. As soon as the manipulated variable $x_a$ decreases, the curve flattens out, and the robot asymptotically approaches the target value of \qty{25}{\cm}. Thanks to the integrating behavior of the open loop (the I-part of its IT1 characteristic), the final value is reached without steady-state error.
                \end{itemize}

                \item \textbf{3. Necessary nonlinear element:} To correctly model the real-world conditions in the simulation, a saturation element must be added. As calculated above, the linear P controller initially outputs values well above \qty{100}{\percent}. The saturation element limits the manipulated variable $x_a$ in the model to the physically possible range $[\qty{-100}{\percent}, \qty{100}{\percent}]$, which corresponds to the actual behavior of the motor driver.
            \end{itemize}

        \section{i) Recorded test-drive vs. simulation}\label{sec:i-}

            \begin{displaycquote}{RTP_Versuche}
                \footnotesize{\textit{(i) Read the recorded data of your test-drive using Serial.ICE and compare it to the simulated results - distance, travelled distance and duty-cycle of motor 0! This has to be done in one graph after importing the data according to the previous experiment!}}
            \end{displaycquote}\label{task(i)}
            
            \begin{figure}[H]
                \centering
                \includegraphics[width=\textwidth]{figure/Comparison_recorded_vs_simulation.pdf}
                \caption{Recorded test-drive (solid) versus simulation (dashed): distance to the obstacle, travelled distance, and motor-0 duty cycle. The simulation is shifted by \qty{-0.5}{\second} onto the recording's time base.}
                \label{fig:compare-recorded-sim}
            \end{figure}

            \paragraph{Comparison and error analysis.}
                Measurement and simulation are nearly identical once the simulation is shifted by \qty{-0.5}{\second} onto the recording's time base. Both reach the setpoint with zero steady-state error (final distance \qty{25.0}{\cm} in either case), which confirms the integral action of the open loop. The controlled variable settles to within \qty{1}{\cm} of the target at \qty{1.61}{\second} (measurement) and \qty{1.64}{\second} (simulation) with a transient RMS deviation of \qty{2.9}{\cm} (maximum \qty{5.8}{\cm} just after the step). The travelled distance ends at \qty{57.2}{\cm} from the encoder odometry against \qty{55.0}{\cm} in the simulation, a deviation of \qty{2.2}{\cm} (\qty{4}{\percent}) that is explained by the slightly larger initial distance of the real run \qty{81}{\cm} vs.\ \qty{80}{\cm} and minor wheel slip. The duty cycle saturates at \qty{100}{\percent} from \qty{0.48}{\second} in both cases; the real motor stays saturated about \qty{0.1}{\second} longer and then drops more abruptly and in discrete steps, reflecting the sampling dead-time and quantisation that the idealized model omits.
            
            \paragraph{Stepped (quantised) curves.}
                Unlike the continuous simulation, the recorded signals take on a visible staircase shape that stems from the digital nature of the measurement chain. The ultrasonic sensor resolves the distance $d$ only in steps of \qty{0.5}{\cm}, so its trace advances in discrete treads instead of a smooth line. The motor duty cycle $\mathrm{Mot0}$ is reported as integer percent (\qty{1}{\percent} resolution), and because the rate limiter only allows a small change per sampling cycle, it climbs as a staircase rather than a ramp. On top of that, the controller runs at a fixed sampling rate of \qty{500}{\hertz}, so every value is held constant between samples (zero-order hold). The wheel encoders, counting up to ${\sim}1170$ steps, are fine enough to appear practically smooth at this scale.

        \section{j) Critical gain and rate-limit-induced oscillation}\label{sec:j-}

            \begin{displaycquote}{RTP_Versuche}
                \footnotesize{\textit{(j) Increase $K_P$, check the influence of the rate-limit-factor in a way that the robot shows periodic oscillations - write down this critical value(s) and upload a video.}}
            \end{displaycquote}\label{task(j)}


            \paragraph{Critical gain.}
                To force the robot into oscillation, the damping ratio is lowered from $D=1$ to $D=0.5$, which quadruples the magnitude of the proportional gain. Evaluating Equation~\ref{eq:K_P} for the right motor gives the critical value
                \begin{equation}
                    K_{P,R} = \frac{-1}{4 \cdot 0.50^2 \cdot 0.0522 \cdot 0.446} = \qty{-42.95}{\percent\per\cm},
                \end{equation}
                and $K_P = -42$ was set on the robot (Listing~\ref{lst:serial-ice-commands-osc}). At this gain the closed loop is strongly underdamped: instead of approaching the \qty{25}{\cm} setpoint aperiodically, the robot overshoots and swings back and forth around it.

            \paragraph{Influence of the rate-limit factor.}
                The rate-limit factor caps how fast the motor duty cycle may change per sampling cycle. Its main purpose is to prevent wheel slip --- and the resulting odometry errors --- when the robot drives at full speed, but it also acts like an additional phase lag (an effective dead-time) in the control loop. Together with the discrete sampling at \qty{500}{\hertz}, this lag reduces the phase margin so far that, at the high gain above, the robot no longer settles but enters a sustained, nearly periodic oscillation (a limit cycle). The oscillation is therefore not perfectly harmonic: its amplitude and period are governed by the dead-time and the rate limiter rather than by the linear $PT_2$ model alone.
